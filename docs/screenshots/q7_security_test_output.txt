INSERT INTO job (job_title) VALUES ('Intruder');                  -- DENIED (read-only)

SELECT e.full_name, s.salary_amount
FROM employee e JOIN job_assignment a USING (employee_id)
JOIN employee_salary s USING (assignment_id)
